Addendum: Add vertical text using iText (C#)

using System;

using iText.IO.Font;

using iText.Kernel.Colors;

using iText.Kernel.Font;

using iText.Kernel.Pdf;

using iText.Layout;

using iText.Layout.Element;

using iText.Layout.Renderer;


namespace iText.Samples.Sandbox.Layout {

    // Example showing how to render text vertically using VerticalParagraph.

    // Demonstrates both VERTICAL_LR and VERTICAL_RL, mixing English and CJK characters.

    public class VerticalText {

        public static readonly string DEST = "results/sandbox/layout/verticalText.pdf";

        public static readonly string FONT = "../../../resources/font/NotoSansCJKjp-Regular.otf";


        public static void Main(String[] args) {

            using (PdfDocument pdfDoc = new PdfDocument(new PdfWriter(DEST))) {

                using (Document doc = new Document(pdfDoc)) {

                    doc.SetFont(PdfFontFactory.CreateFont(FONT, PdfEncodings.IDENTITY_H));


                    // Left-to-right lines writing progression for vertical text.

                    VerticalParagraph ltr = new VerticalParagraph(false);

                    ltr.Add("VERTICAL_LR\n").Add("左から右へ\n").Add("Columns: left to right.\n")

                        .Add("日本語・中文・한국어\n");

                    ltr.SetFontSize(14).SetBackgroundColor(new DeviceRgb(235, 245, 255))

                        .SetFontColor(new DeviceRgb(40, 60, 120)).SetHeight(400).SetMarginRight(30);


                    // Right-to-left lines writing progression — standard for CJK vertical typography.

                    VerticalParagraph rtl = new VerticalParagraph(true);

                    rtl.Add("VERTICAL_RL\n").Add("右から左へ\n").Add("Columns: right to left.\n")

                        .Add("日本語・中文・한국어\n");

                    rtl.SetFontSize(14).SetBackgroundColor(new DeviceRgb(235, 250, 238))

                        .SetFontColor(new DeviceRgb(20, 100, 60)).SetHeight(400);   


                    Div columns = new Div().Add(ltr).Add(rtl);

                    columns.SetNextRenderer(new FlexContainerRenderer(columns));

                    doc.Add(columns);

                }

            }

        }

    }

}
