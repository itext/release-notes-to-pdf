Addendum: Add vertical text using iText (Java)

package com.itextpdf.samples.sandbox.layout;


import com.itextpdf.kernel.colors.DeviceRgb;

import com.itextpdf.kernel.font.PdfFontFactory;

import com.itextpdf.kernel.pdf.PdfDocument;

import com.itextpdf.layout.element.Div;

import com.itextpdf.layout.element.VerticalParagraph;

import com.itextpdf.layout.renderer.FlexContainerRenderer;


/*

 * Example showing how to render text vertically using VerticalParagraph.

 * Demonstrates both VERTICAL_LR and VERTICAL_RL, mixing English and CJK characters.

 */

public class VerticalText {

    public static final String DEST = "./target/sandbox/layout/verticalText.pdf";

    public static final String FONT = "./src/main/resources/font/NotoSansCJKjp-Regular.otf";


    public static void main(String args[]) throws java.io.IOException {

        try (PdfDocument pdfDoc = new PdfDocument(new com.itextpdf.kernel.pdf.PdfWriter(DEST));

             com.itextpdf.layout.Document doc = new com.itextpdf.layout.Document(pdfDoc)) {

            doc.setFont(PdfFontFactory.createFont(FONT, com.itextpdf.io.font.PdfEncodings.IDENTITY_H));


            // Left-to-right lines writing progression for vertical text.

            VerticalParagraph ltr = new VerticalParagraph(false);

            ltr.add("VERTICAL_LR\n").add("左から右へ\n").add("Columns: left to right.\n")

                    .add("日本語・中文・한국어\n");

            ltr.setFontSize(14).setBackgroundColor(new DeviceRgb(235, 245, 255))

                    .setFontColor(new DeviceRgb(40, 60, 120)).setHeight(400).setMarginRight(30);


            // Right-to-left lines writing progression — standard for CJK vertical typography.

            VerticalParagraph rtl = new VerticalParagraph(true);

            rtl.add("VERTICAL_RL\n").add("右から左へ\n").add("Columns: right to left.\n")

                    .add("日本語・中文・한국어\n");

            rtl.setFontSize(14).setBackgroundColor(new DeviceRgb(235, 250, 238))

                    .setFontColor(new DeviceRgb(20, 100, 60)).setHeight(400);


            Div columns = new Div().add(ltr).add(rtl);

            columns.setNextRenderer(new FlexContainerRenderer(columns));

            doc.add(columns);

        }

    }

}
